deploy.ts
// npm i @arvenix/sdk
import { Arvenix } from "@arvenix/sdk";

const arv = new Arvenix({ token: process.env.ARVENIX_TOKEN });

const app = await arv.apps.deploy({
  name:    "checkout-api",
  regions: ["iad", "fra", "sin"],
  scale:   { min: 2, max: 64 },
});

console.log(app.url); // https://checkout-api.arvenix.app
